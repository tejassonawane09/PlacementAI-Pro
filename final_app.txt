# ========== SIMPLE SLIDER FUNCTION - NO DYNAMIC EFFECTS ==========
def create_simple_slider(label, emoji, min_val, max_val, default_val, step, key):
    """Create a clean, minimal slider without dynamic effects"""
    
    # Simple label
    st.markdown(f"**{emoji} {label}**")
    
    # Native Streamlit slider - no effects
    value = st.slider(
        "", 
        min_value=min_val, 
        max_value=max_val, 
        value=default_val, 
        step=step,
        key=key,
        label_visibility="collapsed"
    )
    
    # Show value as plain text
    st.caption(f"Value: {value}")
    st.markdown("---")
    
    return value
# ========== END OF FUNCTION ==========
